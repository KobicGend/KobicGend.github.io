$$
\begin{array}{c|ll|ll|}
& & RC\text{ 电路} & & RL\text{ 电路}
\\ \hline
\text{零状态响应}
& \begin{cases}
R \cdot C \dfrac{\mathrm du_{C}}{\mathrm dt} + u_{C} = U_{\mathrm s} \\ 
u_{C}(0) = 0
\end{cases}
& \Longrightarrow
u_{C} = U_{\mathrm s} \big(1-\mathrm{e}^{-\frac{1}{RC} t}\big) & \begin{cases}
\dfrac{1}{R} \cdot L\dfrac{\mathrm di_{L}}{\mathrm dt} + i_{L} = \dfrac{U_{\mathrm s}}{R} \\ 
i_{L}(0) = 0
\end{cases}
& \Longrightarrow
i_{L} = I_{\mathrm s} \big(1-\mathrm{e}^{-\frac{1}{L / R} t}\big)
\\
\text{零输入响应} 
& \begin{cases}
R \cdot C \dfrac{\mathrm du_{C}}{\mathrm dt} + u_{C} = 0  \\
u_{C}(0) = U_{0}
\end{cases}
& \Longrightarrow
u_{C} = U_{0} \cdot \mathrm{e}^{-\frac{1}{RC} t}
& \begin{cases}
\dfrac{1}{R} \cdot L\dfrac{\mathrm di_{L}}{\mathrm dt} + i_{L} = 0 \\ 
i_{L}(0) = \dfrac{U_{\mathrm 0}}{R}
\end{cases}
& \Longrightarrow
i_{L} = I_{\mathrm 0} \mathrm{e}^{-\frac{1}{L / R} t}
\\
\text{全响应}
& \begin{cases}
R \cdot C \dfrac{\mathrm du_{C}}{\mathrm dt} + u_{C} = U_{\mathrm s} \\
u_{C}(0) = U_{0}
\end{cases}
& \Longrightarrow
u_C = (U_{0} - U_{\mathrm s}) \mathrm{e}^{-\frac{1}{RC} t} + U_{\mathrm s}
& \begin{cases}
\dfrac{1}{R} \cdot L\dfrac{\mathrm di_{L}}{\mathrm dt} + i_{L} = \dfrac{U_{\mathrm s}}{R} \\ 
i_{L}(0) = \dfrac{U_{0}}{R}
\end{cases}
& \Longrightarrow
i_{L} = (I_{0}-I_{\mathrm s}) \mathrm{e}^{-\frac{1}{L / R} t}
+ I_{s}
\\
\end{array}
$$
