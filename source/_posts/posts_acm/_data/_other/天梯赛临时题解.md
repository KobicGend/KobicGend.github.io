## 2020M L1-7 谷歌的招聘 - UPSOLVE

求最早出現的 $k$ 位素數，時間複雜度 $$\Theta(n\sqrt n)$$

賽時 $$18/20$$ 分。需輸出前導零，因此以 `inum` 和 `snum` 分別存整型和字符串型。

```cpp
bool isPrime(int x) {
    for (int j = 2; j * j <= x; ++j)
        if (x % j == 0) return 0;
    return x > 1;
}

void eachT() {
    int n = read(), k = read(), mod = 1;
    int inum = 0;
    string snum;
    
    for (int i = 0; i < k - 1; ++i) {
        char c = getchar();
        inum = (inum << 3) + (inum << 1) + c - '0';
        snum += c;
        mod = (mod << 3) + (mod << 1);
    } // 前k-1個數直接放入
    
    for (int i = k - 1; i < n; ++i) {
        char c = getchar();
        inum = (inum << 3) + (inum << 1) + c - '0';
        snum += c; // 添尾
        
        if (isPrime(inum)) {
            cout << snum << endl;
            return;
        } // 判斷
        
        inum %= mod;
        snum.erase(snum.begin()); // 去頭
    }
    
    cout << 404 << endl;
}
```

方法一：注意到每個數後面比它小的數一定是降序排列的，時間複雜度 $$\Theta(n^2)$$：

```cpp
void eachT() {
    int size = read(), n = read(), q = read();
    while (q--) {
        int mx = 0, flag = 1;

            a[i] = read();

        for (int i = 1; i <= n; ++i) {
            int mn = a[i];

                    if (a[j] < mn) mn = a[j];

                }
            }
        }

    }
}
```

```cpp
void eachT() {
```

```cpp
void eachT() {
    int n = rd, num = n, pos = 0;
    vector<int> a(n), lst(n, -1);

    while (num > 1) {
        for (int i = 0; i < n; ++i) {
            if (a[i]) {
                pos = max(pos + 1, lst[i] + 1);
                ans[i].push_back(pos), --a[i];
                lst[i] = pos;
                if (!a[i]) num--;
            }
        }
    }

    for (int i = 0; i < n; ++i) {
        if (a[i]) pos = max(pos + 1, lst[i] + 2);
        while (a[i]) {
            ans[i].push_back(pos), --a[i];
            pos += 2;
        }
    }

    for (int i = 0; i < n; ++i) {
        printf("#%d\n", i + 1);
        for (int j = 0; j < ans[i].size(); ++j) {

    }
}
```
