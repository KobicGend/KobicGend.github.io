细节题，没有任何思考难度，锻炼码力。

```cpp
const int N = 3e6 + 8;
int a[N], b[N];


bool solve(int l, int r) {
	if (l == -1) return 1;
	int L = a[l], R = a[r];
	if (l == r - 1 && (L >> 1 == R || L == R >> 1)) return 1;
	while (mylg2(L) > mylg2(R)) {
		b[++l] = L >>= 1;
		if (l == r - 1 && (L >> 1 == R || L == R >> 1)) return 1;
		if (l >= r) return 0;
	}
	while (mylg2(L) < mylg2(R)) {
		b[--r] = R >>= 1;
		if (l == r - 1 && (L >> 1 == R || L == R >> 1)) return 1;
		if (l >= r) return 0;
	}

	if ((r - l) & 1) return 0;

	while (L != R) {
		b[++l] = L >>= 1;
		b[--r] = R >>= 1;
		if (L == 0 || R == 0 || l == r && L != R) return 0;
	}

	int op = 1;
	for (int i = l + 1; i <= r - 1; ++i) {
		b[i] = (op + 1) * L;
		op ^= 1;
	}
	return 1;
}

void eachT() {
	int n = read();
	for (int i = 1; i <= n; ++i) {
		b[i] = a[i] = read();
	}

	int lst = -1, now = -1;
	for (int i = 1; i <= n; ++i) {
		if (a[i] != -1) {
			now = i;
			if (!solve(lst, now)) {
				printf("-1\n");
				return;
			}
			lst = now;
		}
	}

	bool flag = 0;

	for (int i = 1; i <= n; ++i) {
		if (a[i] != -1) {
			for (i = i - 1; i >= 1; --i) {
				b[i] = b[i + 1] / 2;
				if (b[i] == 0) b[i] = b[i + 1] << 1;
			}
			flag = 1;
			break;
		}
	}

	for (int i = n; i >= 1; --i) {
		if (a[i] != -1) {
			for (i = i + 1; i <= n; ++i) {
				b[i] = b[i - 1] / 2;
				if (b[i] == 0) b[i] = b[i - 1] << 1;
			}
			flag = 1;
			break;
		}
	}

	if (!flag) {
		b[1] = 1;
		for (int i = 2; i <= n; ++i) {
			b[i] = b[i - 1] / 2;
			if (b[i] == 0) b[i] = b[i - 1] << 1;
		}
	}

	for (int i = 1; i <= n; ++i) {
		printf("%d ", b[i]);
	}
	printf("\n");
}
```

## [CF1930C. Lexicographically Largest](https://codeforces.com/contest/1930/problem/C) (1500)

贪心好题，需要一些思维。

一个长度为 $n$ 的数组 $a$ 。他还有一个空集 $S$ 。注意 $S$ 不是一个可重集。

他将进行以下三步操作恰好 $n$ 次：选择一个 $i$，将 $a_i + i$ 插入 $S$，并从 $a$ 中删除 $a_i$。注意，$a_i$ 右边所有元素的索引将减少 $1$。

现在，Willow 将构造一个新数组 $b$ ，这个数组是 $S$ 按递减顺序排列，希望最后 $b$ 的字典序最大。

```cpp
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
 
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
 
    int t;
    cin >> t;

    while (t--) {
        int n;
        cin >> n;
        
        vector<int> a(n);
        for (int i = 0; i < n; i++) {
            cin >> a[i];
            a[i] += i + 1;
        }
        sort(a.begin(), a.end(), greater<>());

        for (int i = 0; i < n; i++) {
            if (i) a[i] = min(a[i], a[i - 1] - 1);
            cout << a[i] << " \n"[i == n - 1];
        }
    }
 
    return 0;
}
```
